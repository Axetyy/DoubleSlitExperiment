407def refresh():
408 with torch.no_grad():
409 field = sim.u[
410 crop_z[0]:crop_z[1]:RENDER_STRIDE,
411 crop_y[0]:crop_y[1]:RENDER_STRIDE,
412 :visible_x:RENDER_STRIDE,
413 ]
414 first_wave_x = (p.x_src + 4) // RENDER_STRIDE
415 local_peak = field.abs().amax(dim=(0, 1), keepdim=True).clamp_min(1e-7)
416 reference_peak = field[:, :, first_wave_x:].abs().amax()
417 active_slices = local_peak > reference_peak * 0.001
418 positive_peak = field.clamp_min(0).amax(dim=(0, 1), keepdim=True)
419 negative_peak = (-field.clamp_max(0)).amax(dim=(0, 1), keepdim=True)
420 positive_scale = torch.where(positive_peak > local_peak * 0.02,
421 positive_peak, local_peak)
422 negative_scale = torch.where(negative_peak > local_peak * 0.02,
423 negative_peak, local_peak)
424 values = torch.where(
425 field >= 0,
426 field / positive_scale,
427 field / negative_scale,
428 )
429 values = torch.where(active_slices, values, 0.0).clamp(-1.0, 1.0)
430 values[:, :, :first_wave_x] = 0
431 values = values.contiguous().cpu().numpy().ravel(order="C")
432
433 grid.point_data["wave"][:] = values
434 wave_scalars.Modified()
435 grid.Modified()
436 negative_contour.Modified()
437 positive_contour.Modified()
438 negative_actor.SetVisibility(state["wavefronts"])
439 positive_actor.SetVisibility(state["wavefronts"])
440
441 detector_image = sim.detector_image()
442 update_detector(detector_image)
443 if state["particles"]:
444 profile = sampler.hist.sum(axis=1).astype(np.float64)
445 else:
446 profile = detector_image[:, p.nz // 2]
447 profile_peak = float(profile.max())
448 normalized_profile = profile / max(profile_peak, 1e-12)
449 measured_line.update(screen_y_mm, normalized_profile)
450
451 if sim.t_step < arrival_step:
452 message = f"wave travelling to detector: {sim.t_step}/{arrival_step} steps"
453 elif state["particles"]:
454 message = f"detector accumulated {int(sampler.hist.sum())} hits"
455 else:
456 message = f"detector intensity peak {float(sim.detector_image().max()):.3g}"
457 if state["paused"]:
458 message = "paused | " + message
459 status_actor.SetInput(f"{sim.label}\n{message}")
460 status_actor.Modified()
