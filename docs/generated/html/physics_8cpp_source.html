192void Tensor2D::laplacianInto(Tensor2D &out) const
193{
194 requireSameShape(out, "laplacianInto");
195 if (rows_ == 0 || cols_ == 0)
196 return;
197
198 out.d_[0].assign(cols_, 0.0f);
199 out.d_[rows_ - 1].assign(cols_, 0.0f);
200
201#ifdef _OPENMP
202#pragma omp parallel for schedule(static)
203#endif
204 for (int y = 1; y < rows_ - 1; ++y)
205 {
206 const float *c = d_[y].data();
207 const float *up = d_[y - 1].data();
208 const float *dn = d_[y + 1].data();
209 float *o = out.d_[y].data();
210 o[0] = 0.0f;
211 o[cols_ - 1] = 0.0f;
212 for (int x = 1; x < cols_ - 1; ++x)
213 o[x] = up[x] + dn[x] + c[x - 1] + c[x + 1] - 4.0f * c[x];
214 }
215}

221void Tensor2D::rowMeanSq(int x0, int x1, std::vector<float> &out) const
222{
223 x0 = std::max(0, x0);
224 x1 = std::min(cols_ - 1, x1);
225 if (x0 > x1)
226 throw std::invalid_argument("rowMeanSq: empty column range");
227 out.assign(rows_, 0.0f);
228 const float inv = 1.0f / static_cast<float>(x1 - x0 + 1);
229 for (int y = 0; y < rows_; ++y)
230 {
231 const float *r = d_[y].data();
232 float s = 0.0f;
233 for (int x = x0; x <= x1; ++x)
234 s += r[x] * r[x];
235 out[y] = s * inv;
236 }

273void WaveSimulator::buildAbsorber()
274{
275 const int nx = p_.nx, ny = p_.ny;
276 const double sp = p_.sponge;
277
278 std::vector<float> depth_x(nx), depth_y(ny);
279 for (int x = 0; x < nx; ++x)
280 {
281 depth_x[x] = static_cast<float>(
282 std::max(0.0, sp - std::min(x, nx - 1 - x)) / sp);
283 }
284 for (int y = 0; y < ny; ++y)
285 {
286 depth_y[y] = static_cast<float>(
287 std::max(0.0, sp - std::min(y, ny - 1 - y)) / sp);
288 }
289
290 dampening_prev_ = Tensor2D(ny, nx);
291 dampening_next_ = Tensor2D(ny, nx);
292
293 for (int y = 0; y < ny; ++y)
294 {
295 for (int x = 0; x < nx; ++x)
296 {
297 const float e = std::max(depth_y[y], depth_x[x]);
298 const float s = 0.6f * e * e;
299 dampening_prev_(y, x) = 1.0f - 0.5f * s;
300 dampening_next_(y, x) = 1.0f / (1.0f + 0.5f * s);
301 }
302 }

306void WaveSimulator::rebuild(bool fresh)
307{
308 const int nx = p_.nx, ny = p_.ny;
309 const double yc = ny / 2.0;
310
311 // Slit centres / open rows
312 slit_centers_.resize(p_.n_slits);
313 for (int i = 0; i < p_.n_slits; ++i)
314 slit_centers_[i] = yc + (i - (p_.n_slits - 1) / 2.0) * p_.slit_spacing;
315
316 slit_open_.assign(ny, 0);
317 for (int y = 0; y < ny; ++y)
318 for (double c : slit_centers_)
319 if (std::fabs(y - c) <= p_.slit_width / 2.0)
320 {
321 slit_open_[y] = 1;
322 break;
323 }
324
325 // Wall mask: 1 = open, 0 = solid
326 wall_mask_ = Tensor2D(ny, nx, 1.0f);
327 const int xw0 = std::max(0, p_.x_wall);
328 const int xw1 = p_.x_wall + std::min(p_.wall_thickness, nx - p_.x_wall);
329 for (int y = 0; y < ny; ++y)
330 for (int x = xw0; x < xw1; ++x)
331 wall_mask_(y, x) = slit_open_[y] ? 1.0f : 0.0f;
332
333 // Source position and phase
334 const double tan_t = std::tan(math_stuff::deg2rad(p_.source_angle_deg));
335 double yc_src = yc - (p_.x_wall - p_.x_src) * tan_t;
336 yc_src = std::min(std::max(yc_src, static_cast<double>(p_.sponge)),
337 static_cast<double>(ny - p_.sponge - 1));
338 src_y_ = static_cast<int>(std::nearbyint(yc_src));
339
340 const double k = 2.0 * math_stuff::kPi / p_.wavelength;
341 phase_offset_ = -k * (src_y_ - yc_src) * std::sin(math_stuff::deg2rad(p_.source_angle_deg));
342 phase_noise_ = 0.0;
343 omega_ = 2.0 * math_stuff::kPi / p_.wavelength;
344
345 c_next_ = dampening_next_ * wall_mask_;
346 c_prev_ = dampening_prev_ * dampening_next_ * wall_mask_;
347 src_mask_ = wall_mask_(src_y_, p_.x_src);
348
349 const double propagation_steps = (p_.x_det - p_.x_src) / static_cast<double>(p_.courant);
350 settle_ = static_cast<int>(propagation_steps + 2 * AVG_STEPS);
351 if (fresh)
352 settle_ += static_cast<int>((p_.x_wall - p_.x_src) / static_cast<double>(p_.courant));
353}

369float WaveSimulator::sourceValue(long long t)
370{
371 double phase = omega_ * p_.courant * static_cast<double>(t) + phase_offset_;
372 if (!p_.coherent)
373 {
374 phase_noise_ += 0.1 * normal_(rng_);
375 phase += phase_noise_;
376 }
377 const double amp = std::min(1.0, static_cast<double>(t) / 100.0);
378 return static_cast<float>(amp * std::sin(phase) * src_mask_);
379}

383void WaveSimulator::step_n(int n)
384{
385 if (n <= 0)
386 return;
387
388 const float C2 = p_.courant * p_.courant;
389 const int x0 = std::max(0, p_.x_det - p_.detector_width + 1);
390 const int x1 = p_.x_det;
391 const float inv_avg = 1.0f / static_cast<float>(AVG_STEPS);
392
393 for (int k = 0; k < n; ++k)
394 {
395 // un = cn * (2u + C2 * lap(u)) - cp * up (lap = 0 on the border)
396 u_next_.assign(u_);
397 u_next_ *= 2.0f;
398 u_.laplacianInto(lap_);
399 lap_ *= C2;
400 u_next_ += lap_;
401 u_next_ *= c_next_;
402 u_next_.addcmul(c_prev_, u_prev_, -1.0f);
403
404 u_next_(src_y_, p_.x_src) += sourceValue(t_step_ + k);
405
406 u_next_.rowMeanSq(x0, x1, inst_);
407 for (int y = 0; y < p_.ny; ++y)
408 I_det_[y] += (inst_[y] - I_det_[y]) * inv_avg;
409
410 // rotate: u -> u_next -> u_prev -> u (O(1))
411 u_prev_.swap(u_);
412 u_.swap(u_next_);
413 }
414
415 t_step_ += n;
416 settle_ = std::max(0, settle_ - n);
417}
