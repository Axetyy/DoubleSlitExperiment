1"""
2Main script for the double-slit experiment.
3
4This is where you run the simulation and visualize the results. It uses the parameters defined in config.py to set up the simulation, and then runs the simulation step by step, updating the matplotlib figures to show the wave field propagating. It is through using this script that you can interact with the simulation, changing parameters and observing how the wave field and detector readings change in response.
5
6The simulation can be run in 2D or 3D, and can use different backends for the wave simulation including:
7 - Numba ( CPU ), a powerful just-in-time compiler for Python that can speed up numerical computations by compiling python to machine code at runtime.
8 - C++ ( CPU )
9 - PyTorch ( CPU or GPU ), a popular machine learning library that can also be used for general numerical computations, and can leverage the power of GPUs for faster computation.
10
11The simulation can also be run in particle mode, where a sampler simulates the detection of indvidual photons at the detector, and the results are compared to the expected Fraunhofer diffraction pattern. ( Monte Carlo simulation of photon detection. )
12

267def build_sim():
268 """
269 Build the simulation object based on the current parameters and backend selection.
270 """
271 global sim, sampler, backend_label, DEV
272 dev = DEV
273 if dev == "GPU" and not cuda_available():
274 say("CUDA GPU not available - running on CPU instead.")
275 dev = "CPU"
276 DEV = "CPU"
277 radio_quiet(rb_dev, 0)
278 else:
279 say("")
280
281 if DIM == 2:
282 if dev == "GPU":
283 sim = TorchWaveSim2D(p, "cuda")
284 elif dev == "C++":
285 if CppWaveSim2D is None:
286 raise RuntimeError("C++ backend unavailable.")
287 sim = CppWaveSim2D(p)
288 elif NumbaSim2D is not None:
289 sim = NumbaSim2D(p)
290 sim.label = "CPU numba"
291 elif TorchWaveSim2D is not None:
292 sim = TorchWaveSim2D(p, "cpu")
293 else:
294 raise RuntimeError("Install numba or torch.")
295 sampler = ParticleSampler(p.ny, p.x_det)
296 else:
297 if physics3d is None:
298 raise RuntimeError("physics3d unavailable.")
299 if dev == "GPU":
300 sim = physics3d.TorchWaveSim3D(p, "cuda")
301 elif physics3d.HAVE_NUMBA:
302 sim = physics3d.WaveSim3D(p)
303 else:
304 sim = physics3d.TorchWaveSim3D(p, "cpu")
305 sampler = ParticleSampler3D(p.ny, p.nz, p.x_det)
306
307 backend_label = f"{DIM}D | {getattr(sim, 'label', 'CPU')}"

485def draw_3d(disp):
486 """
487 Draw the 3D surface.
488 """
489 global slice_surf, screen_surf
490 cmap = plt.get_cmap("coolwarm")
491 xs = min(p.nx, p.x_det + 9)

506 else:
507 img = det_image()

509 colors = np.ones(img.shape + (4,))
510 colors[..., :3] = img[..., None] * screen_color[None, None, :]
511 colors = colors[::SURF_STEP, ::SURF_STEP]
512 yi = np.arange(0, p.ny, SURF_STEP)[: colors.shape[0]]
513 zi = np.arange(0, p.nz, SURF_STEP)[: colors.shape[1]]
514 Yg, Zg = np.meshgrid(yi, zi, indexing="ij")
515 Xg = np.full(Yg.shape, float(p.x_det))
516 if screen_surf is not None:
517 screen_surf.remove()
518 screen_surf = ax_f3.plot_surface(Xg, Yg, Zg, facecolors=colors, rstride=1, cstride=1,
519 shade=False, linewidth=0, antialiased=False)
