42@njit(parallel=PARALLEL, cache=False)
43def _one_step(u:np.ndarray, up:np.ndarray, un:np.ndarray, c_next:np.ndarray, c_prev:np.ndarray, C2:np.float32, sy:int, sx:int, sv:np.float32, x0:int, x1:int, I_det:np.ndarray, avg:np.float32):
44 """One leapfrog step, fused.
45
46 un = mask * ( dn * (2u + C2*lap(u)) - dn*dp * up + source )
47 with c_next = dn*mask and c_prev = dn*dp*mask precomputed.
48 Border cells have lap == 0
49
50 :param u: current field
51 :param up: previous field
52 :param un: next field (output)
53 :param c_next: precomputed damping * mask for next field
54 :param c_prev: precomputed damping * damping_prev * mask for previous field
55 :param C2: courant^2
56 :param sy: y-coordinate of the source
57 :param sx: x-coordinate of the source
58 :param sv: source value to inject
59 :param x0: left edge of the detector
60 :param x1: right edge of the detector
61 :param I_det: detector intensity accumulator (output)
62 :param avg: number of steps to average over for the detector
63
64 :return None: modifies un and I_det in place
65 """
66 ny, nx = u.shape
67 two = np.float32(2.0)
68 four = np.float32(4.0)
69 inv_w = 1.0 / (x1 - x0 + 1)
70
71 for i in prange(ny):
72 if i == 0 or i == ny - 1:
73 for j in range(nx):
74 un[i, j] = c_next[i, j] * (two * u[i, j]) - c_prev[i, j] * up[i, j]
75 else:
76 un[i, 0] = c_next[i, 0] * (two * u[i, 0]) - c_prev[i, 0] * up[i, 0]
77 un[i, nx - 1] = (
78 c_next[i, nx - 1] * (two * u[i, nx - 1])
79 - c_prev[i, nx - 1] * up[i, nx - 1]
80 )
81 for j in range(1, nx - 1):
82 lap = C2 * (
83 u[i + 1, j] + u[i - 1, j] + u[i, j + 1] + u[i, j - 1]
84 - four * u[i, j]
85 )
86 un[i, j] = c_next[i, j] * (two * u[i, j] + lap) - c_prev[i, j] * up[i, j]
87
88 if i == sy:
89 un[i, sx] += sv
90
91 # detector: running mean of u^2 over the last columns, float64 accumulator
92 s = 0.0
93 for j in range(x0, x1 + 1):
94 v = un[i, j]
95 s += v * v
96 I_det[i] += np.float32((s * inv_w - I_det[i]) / avg)
97

99@njit(cache=False)
100def _run_steps(u:np.ndarray, up:np.ndarray, un:np.ndarray, c_next:np.ndarray, c_prev:np.ndarray, C2:np.float32, sy:int, sx:int, sv_arr:np.ndarray, x0:int, x1:int, I_det:np.ndarray, avg:np.float32):
101 """Run len(sv_arr) steps, rotating the three buffers. Returns (u, u_prev, u_next)."""
102 for k in range(sv_arr.shape[0]):
103 _one_step(u, up, un, c_next, c_prev, C2, sy, sx, sv_arr[k], x0, x1, I_det, avg)
104 u, up, un = un, u, up
105 return u, up, un

230 def step_n(self, n=1):
231 """Advance n time steps in a single compiled call."""
232 if n <= 0:
233 return
234
235 p = self.p
236
237 t = self.t_step + np.arange(n)
238 phase = self.omega * p.courant * t + self.phase_offset[0]
239
240 if not p.coherent:
241 noise = self.phase_noise[0] + np.cumsum(0.1 * self.rng.standard_normal(n))
242 self.phase_noise[0] = noise[-1]
243 phase = phase + noise
244
245 amplitude = np.minimum(1.0, t / 100)
246 sv = (amplitude * np.sin(phase) * self.src_mask).astype(np.float32)
247
248 x0 = max(0, p.x_det - p.detector_width + 1)
249
250 ny, nx = self.u.shape
251 if not (0 <= int(self.src_y[0]) < ny and 0 <= p.x_src < nx and 0 <= p.x_det < nx):
252 raise ValueError(
253 f"index outside grid: src=({self.src_y[0]},{p.x_src}) x_det={p.x_det} grid=({ny},{nx})"
254 )
255 for a in (self.u, self.u_prev, self.u_next, self.c_next, self.c_prev):
256 if a.shape != (ny, nx) or a.dtype != np.float32 or not a.flags.c_contiguous:
257 raise ValueError("simulation arrays must be C-contiguous float32 of identical shape")
258
259 self.u, self.u_prev, self.u_next = _run_steps(
260 self.u, self.u_prev, self.u_next,
261 self.c_next, self.c_prev, self.C2,
262 int(self.src_y[0]), int(p.x_src), sv,
263 int(x0), int(p.x_det),
264 self.I_det, float(self.AVG_STEPS),
265 )
266
267 self.t_step += n
268 self.settle -= n

271 self.step_n(1)

273 return self.I_det

383 def step_n(self, n=1):
384 """Advance n time steps in a single compiled call."""
385 if n <= 0:
386 return 0
387 p = self.p
388 source_values = torch.from_numpy(self._source_values(n)).to(self.device)
389 u, u_prev, u_next = self.u, self.u_prev, self.u_next
390 x0 = max(0, p.x_det - p.detector_width + 1)
391 C2 = float(p.courant ** 2)
392 sy, sx = int(self.src_y[0]), int(p.x_src)
393 inv_avg = 1.0 / self.AVG_STEPS
394 with torch.no_grad():
395 for k in range(n):
396 torch.mul(u, 2.0, out=u_next)
397 lap = u[2:, 1:-1] + u[:-2, 1:-1]
398 lap += u[1:-1, 2:]
399 lap += u[1:-1, :-2]
400 lap.sub_(u[1:-1, 1:-1], alpha=4.0)
401 u_next[1:-1, 1:-1].add_(lap, alpha=C2)
402 u_next.mul_(self.c_next)
403 u_next.addcmul_(self.c_prev, u_prev, value=-1.0)
404 u_next[sy, sx].add_(source_values[k])
405 instance = u_next[:, x0:p.x_det + 1].pow(2).mean(dim=1)
406 self.I_det += (instance - self.I_det) * inv_avg
407 u, u_prev, u_next = u_next, u, u_prev
408 self.u, self.u_prev, self.u_next = u, u_prev, u_next
409 self.t_step += n
410 self.settle -= n

425@njit(parallel=PARALLEL, cache=False)
426def _one_step_3d(u, up, un, c_next, c_prev, C2, sz, sy, sx, sv, x0, x1, I_det, avg):
427 """One leapfrog step in 3D, fused.
428 un = mask * ( dn * (2u + C2*lap(u)) - dn*dp * up + source )
429 with c_next = dn*mask and c_prev = dn*dp*mask precomputed
430
431 :param u: current field
432 :param up: previous field
433 :param un: next field (output)
434 :param c_next: precomputed damping * mask for next field
435 :param c_prev: precomputed damping * damping_prev * mask for previous field
436 :param C2: courant^2
437 :param sz: z-coordinate of the source
438 :param sy: y-coordinate of the source
439 :param sx: x-coordinate of the source
440 :param sv: source value to inject
441 :param x0: left edge of the detector
442 :param x1: right edge of the detector
443 :param I_det: detector intensity accumulator (output)
444 :param avg: number of steps to average over for the detector
445
446 :return None: modifies un and I_det in place
447 """
448 nz, ny, nx = u.shape
449 two = np.float32(2.0)
450 six = np.float32(6.0)
451 inv_w = 1.0 / (x1 - x0 + 1)
452 for k in prange(nz):
453 for i in range(ny):
454 if k == 0 or k == nz - 1 or i == 0 or i == ny - 1:
455 for j in range(nx):
456 un[k, i, j] = c_next[k, i, j] * (two * u[k, i, j]) - c_prev[k, i, j] * up[k, i, j]
457 else:
458 un[k, i, 0] = c_next[k, i, 0] * (two * u[k, i, 0]) - c_prev[k, i, 0] * up[k, i, 0]
459 un[k, i, nx - 1] = (
460 c_next[k, i, nx - 1] * (two * u[k, i, nx - 1])
461 - c_prev[k, i, nx - 1] * up[k, i, nx - 1]
462 )
463 for j in range(1, nx - 1):
464 lap = C2 * (
465 u[k + 1, i, j] + u[k - 1, i, j]
466 + u[k, i + 1, j] + u[k, i - 1, j]
467 + u[k, i, j + 1] + u[k, i, j - 1]
468 - six * u[k, i, j]
469 )
470 un[k, i, j] = c_next[k, i, j] * (two * u[k, i, j] + lap) - c_prev[k, i, j] * up[k, i, j]
471 if k == sz and i == sy:
472 un[k, i, sx] += sv
473 s = 0.0
474 for j in range(x0, x1 + 1):
475 value = un[k, i, j]
476 s += value * value
477 I_det[i, k] += np.float32((s * inv_w - I_det[i, k]) / avg)
478

520 def _build_damping(self):
521 """Builds a damping profile for the sponge boundary conditions in 3D."""
522 p = self.p
523
524 def depth(n):
525 i = np.arange(n)
526 return (np.maximum(0, p.sponge - np.minimum(i, n - 1 - i)) / p.sponge).astype(np.float32)
527
528 dz = depth(p.nz)[:, None, None]
529 dy = depth(p.ny)[None, :, None]
530 dx = depth(p.nx)[None, None, :]
531 damping = (0.6 * np.maximum(np.maximum(dz, dy), dx) ** 2).astype(np.float32)
532 self.dprev = (1 - 0.5 * damping).astype(np.float32)
533 self.dnext = (1 / (1 + 0.5 * damping)).astype(np.float32)

535 def rebuild(self, fresh=False):
536 """Rebuilds the simulation state, including the wall mask and source position. If fresh is True, the simulation is reset to its initial state."""
537 p = self.p
538 yc, zc = p.ny / 2, p.nz / 2
539 y, z = np.arange(p.ny), np.arange(p.nz)
540 self.slit_centers = (
541 yc + (np.arange(p.n_slits) - (p.n_slits - 1) / 2) * p.slit_spacing
542 )
543 open_y = np.zeros(p.ny, dtype=bool)
544 for center in self.slit_centers:
545 open_y |= np.abs(y - center) <= p.slit_width / 2
546 open_z = np.abs(z - zc) <= p.slit_height / 2
547 self.slit_open = open_y
548 self.slit_open_z = open_z
549 aperture = open_z[:, None] & open_y[None, :]
550 if p.slit_shape == "circular":
551 aperture = np.zeros((p.nz, p.ny), dtype=bool)
552 radius_squared = (p.slit_width / 2) ** 2
553 for center in self.slit_centers:
554 aperture |= (
555 (z[:, None] - zc) ** 2 + (y[None, :] - center) ** 2
556 <= radius_squared
557 )
558 self.aperture_mask = aperture
559 mask = np.ones((p.nz, p.ny, p.nx), dtype=np.float32)
560 mask[:, :, p.x_wall:p.x_wall + p.wall_thickness] = aperture[:, :, None]
561 angle = np.tan(np.deg2rad(p.source_angle_deg))
562 source_y = np.clip(
563 yc - (p.x_wall - p.x_src) * angle,
564 p.sponge,
565 p.ny - p.sponge - 1,
566 )
567 self.src_y = np.array([int(round(source_y))])
568 self.src_z = int(p.nz // 2)
569 k = 2 * np.pi / p.wavelength
570 self.phase_offset = (
571 -k * (self.src_y - source_y) * np.sin(np.deg2rad(p.source_angle_deg))
572 )
573 self.phase_noise = 0.0
574 self.omega = 2 * np.pi / p.wavelength
575 self.src_mask = float(mask[self.src_z, self.src_y[0], p.x_src])
576 self._upload_coeffs(self.dnext * mask, self.dprev * self.dnext * mask)
577 self.settle = int((p.x_det - p.x_src) / p.courant + 2 * self.AVG_STEPS)
578 if fresh:
579 self.settle += int((p.x_wall - p.x_src) / p.courant)
580
581 def reset(self):
582 """Resets the simulation to its initial state."""
583 self._zero_state()
584 self.t_step = 0
585 self.rebuild(fresh=True)
586
587 def _source_values(self, n):
588 """Returns the source values for n time steps, including phase noise if incoherent."""
589 p = self.p
590 t = self.t_step + np.arange(n)
591 phase = self.omega * p.courant * t + self.phase_offset
592 if not p.coherent:
593 noise = self.phase_noise + np.cumsum(0.1 * self.rng.standard_normal(n))
594 self.phase_noise = float(noise[-1])
595 phase = phase + noise
596 amplitude = np.minimum(1.0, t / 100)
597 return (amplitude * np.sin(phase) * self.src_mask).astype(np.float32)
598
599 def step_n(self, n=1):
600 if n <= 0:
601 return 0
602 self._advance(self._source_values(n))
603 self.t_step += n
604 self.settle -= n
605
606 def step(self):
607 self.step_n(1)

691 def _upload_coeffs(self, cn, cp):
692 """Uploads the simulation coefficients to the device."""
693 self.cn = torch.from_numpy(np.ascontiguousarray(cn, dtype=np.float32)).to(self.device)
694 self.cp = torch.from_numpy(np.ascontiguousarray(cp, dtype=np.float32)).to(self.device)

696 def _advance(self, sv_np):
697 """Advance the simulation by one time step with the given source values."""
698 p = self.p
699 sv = torch.from_numpy(sv_np).to(self.device)
700 u, up, un = self.u, self.u_prev, self.u_next
701 C2 = float(p.courant ** 2)
702 sz, sy, sx = int(self.src_z), int(self.src_y[0]), int(p.x_src)
703 x0, x1 = max(0, p.x_det - p.detector_width + 1), p.x_det
704 inv_avg = 1.0 / self.AVG_STEPS
705 with torch.no_grad():
706 for n in range(sv.shape[0]):
707 torch.mul(u, 2.0, out=un)
708 lap = u[2:, 1:-1, 1:-1] + u[:-2, 1:-1, 1:-1]
709 lap += u[1:-1, 2:, 1:-1]
710 lap += u[1:-1, :-2, 1:-1]
711 lap += u[1:-1, 1:-1, 2:]
712 lap += u[1:-1, 1:-1, :-2]
713 lap.sub_(u[1:-1, 1:-1, 1:-1], alpha=6.0)
714 un[1:-1, 1:-1, 1:-1].add_(lap, alpha=C2)
715 un.mul_(self.cn)
716 un.addcmul_(self.cp, up, value=-1.0)
717 un[sz, sy, sx].add_(sv[n])
718 instance = un[:, :, x0:x1 + 1].pow(2).mean(dim=2)
719 self.I_det += (instance.T - self.I_det) * inv_avg
720 u, up, un = un, u, up
721 self.u, self.u_prev, self.u_next = u, up, un
