5def _numerical_wavenumber(p: Params, angle:float)->np.ndarray:
6 """
7 Lattice wavenumber along a direction with angle ( radians ) from the x-asis. Solves sin^2(w dt/2)/C^2 = sin^2(k cos(angle)/2) + sin^2(k sin(angle)/2).
8
9 :param p: simulation parameters
10 :param angle: angle from the x-axis in radians
11 :return: numerical wavenumber
12
13 """
14
15 omega_dt = 2 * np.pi * p.courant / p.wavelength
16 target = np.sin(omega_dt / 2) ** 2 / p.courant ** 2
17 angle = np.asarray(angle, dtype=float)
18 low = np.zeros_like(angle, dtype=float)
19 high = np.full_like(angle, np.pi, dtype=float)
20
21 for _ in range(32):
22 wave_number = (low + high) / 2
23 dispersion = (
24 np.sin(wave_number * np.cos(angle) / 2) ** 2
25 + np.sin(wave_number * np.sin(angle) / 2) ** 2
26 )
27 below_target = dispersion < target
28 low = np.where(below_target, wave_number, low)
29 high = np.where(below_target, high, wave_number)
30
31 return (low + high) / 2

34def fraunhofer(p: Params, detector_rows:np.ndarray, slit_open:np.ndarray, source_row:float)->np.ndarray:
35 """
36 Calculates the fraunhofer diffraction pattern for a set of given parameters
37
38 :param p: simulation parameters
39 :param detector_rows: rows of the detector
40 :param slit_open: boolean array indicating which rows of the slit are open
41 :param source_row: row of the source
42
43 :return: normalized intensity along the detector rows
44 """
45
46 center = p.ny / 2
47 detector_y = np.asarray(detector_rows, dtype=float) - center
48
49 idx = np.flatnonzero(np.asarray(slit_open, dtype=bool))
50 aperture_y = idx.astype(float) - center
51
52 source_x = p.x_wall + p.wall_thickness / 2 - p.x_src
53 screen_x = p.x_det - p.x_wall - p.wall_thickness / 2
54 source_y = float(source_row) - center
55
56 incident_angle = np.arctan2(aperture_y - source_y, source_x)
57 incident_k = _numerical_wavenumber(p, incident_angle)
58 source_distance = np.hypot(source_x, aperture_y - source_y)
59 aperture_field = np.exp(1j * incident_k * source_distance) / np.sqrt(source_distance)
60
61 sin_theta = detector_y / np.hypot(detector_y, screen_x)
62 detector_angle = np.arctan2(detector_y, screen_x)
63 detector_k = _numerical_wavenumber(p, detector_angle)
64 propagation = np.exp(
65 -1j * detector_k[:, None] * sin_theta[:, None] * aperture_y[None, :]
66 )
67 intensity = np.abs(propagation @ aperture_field) ** 2
68
69 peak = float(np.max(intensity)) if intensity.size else 0.0
70 return intensity / peak if peak > 0 else intensity

73def _numerical_wavenumber_3d(p:Params, cx:np.ndarray, cy:np.ndarray, cz:np.ndarray)->np.ndarray:
74 """Lattice wavenumber along a direction with cosines (cx, cy, cz).
75
76 Solves sin^2(w dt/2)/C^2 = sin^2(k cx/2) + sin^2(k cy/2) + sin^2(k cz/2).
77
78 :param p: simulation parameters
79 :param cx : cosine of the angle in the x-direction
80 :param cy : cosine of the angle in the y-direction
81 :param cz : cosine of the angle in the z-direction
82
83 :return: numerical wavenumber
84 """
85 omega_dt = 2 * np.pi * p.courant / p.wavelength
86 target = np.sin(omega_dt / 2) ** 2 / p.courant ** 2
87 cx, cy, cz = np.broadcast_arrays(
88 np.asarray(cx, float), np.asarray(cy, float), np.asarray(cz, float)
89 )
90 low = np.zeros_like(cx)
91 high = np.full_like(cx, np.pi)
92
93 for _ in range(32):
94 k = (low + high) / 2
95 f = (
96 np.sin(k * cx / 2) ** 2
97 + np.sin(k * cy / 2) ** 2
98 + np.sin(k * cz / 2) ** 2
99 )
100 below = f < target
101 low = np.where(below, k, low)
102 high = np.where(below, high, k)
103
104 return (low + high) / 2

107def fraunhofer3d(
108 p:Params, detector_rows:np.ndarray, slit_open_y:np.ndarray, slit_open_z:np.ndarray, source_y:float, source_z:float, aperture_mask=None
109):
110 """
111 Calculates the fraunhofer diffraction pattern for a set of given parameters in 3D.
112
113 :param p: simulation parameters
114 :param detector_rows: rows of the detector
115 :param slit_open_y: boolean array indicating which rows of the slit are open in the y-direction
116 :param slit_open_z: boolean array indicating which rows of the slit are open in the z-direction
117 :param source_y: y-coordinate of the source
118 :param source_z: z-coordinate of the source
119
120 :return: normalized intensity along the detector rows
121 """
122 cy0, cz0 = p.ny / 2, p.nz / 2
123 det_y = np.asarray(detector_rows, dtype=float) - cy0
124
125 if aperture_mask is None:
126 aperture = (
127 np.asarray(slit_open_z, dtype=bool)[:, None]
128 & np.asarray(slit_open_y, dtype=bool)[None, :]
129 )
130 else:
131 aperture = np.asarray(aperture_mask, dtype=bool)
132 iz, iy = np.nonzero(aperture)
133 if iy.size == 0:
134 return np.zeros_like(det_y)
135
136 ay_all = np.unique(iy).astype(float) - cy0
137 aperture_y = iy.astype(float) - cy0
138 aperture_z = iz.astype(float) - cz0
139
140 sx = p.x_wall + p.wall_thickness / 2 - p.x_src
141 scr = p.x_det - p.x_wall - p.wall_thickness / 2
142 sy = float(source_y) - cy0
143 sz = float(source_z) - cz0
144
145 dy = aperture_y - sy
146 dz = aperture_z - sz
147 r = np.sqrt(sx ** 2 + dy ** 2 + dz ** 2)
148 k_in = _numerical_wavenumber_3d(p, sx / r, dy / r, dz / r)
149 cell_fields = np.exp(1j * k_in * r) / r
150 summed_field = np.bincount(iy, weights=cell_fields.real, minlength=p.ny)
151 summed_field = summed_field + 1j * np.bincount(
152 iy, weights=cell_fields.imag, minlength=p.ny
153 )
154 aperture_field = summed_field[np.unique(iy)]
155
156 R = np.hypot(det_y, scr)
157 sin_t = det_y / R
158 k_out = _numerical_wavenumber_3d(p, scr / R, det_y / R, np.zeros_like(R))
159 propagation = np.exp(-1j * k_out[:, None] * sin_t[:, None] * ay_all[None, :])
160 intensity = np.abs(propagation @ aperture_field) ** 2
161
162 peak = float(np.max(intensity)) if intensity.size else 0.0
163 return intensity / peak if peak > 0 else intensity
