137// Re-apply params (except nx/ny/sponge, which are fixed at build) and restart.
138API int sim_reset(void *h)
139{
140 if (!valid(h))
141 return -1;
142 Handle *s = H(h);
143 if (!s->sim)
144 {
145 last_error = "simulator has not been built";
146 return -1;
147 }
148 try
149 {
150 Params np = s->p;
151 const Params cur = s->sim->params();
152 np.nx = cur.nx;
153 np.ny = cur.ny;
154 np.sponge = cur.sponge;
155 s->sim->params() = np;
156 try
157 {
158 s->sim->reset();
159 }
160 catch (...)
161 {
162 s->sim->params() = cur;
163 throw;
164 }
165 last_error.clear();
166 return 0;
167 }
168 catch (const std::exception &e)
169 {
170 return fail(e);
171 }
172}

201API int sim_copy_field(void *h, float *out)
202{
203 if (!valid(h) || !H(h)->sim || !out)
204 {
205 last_error = "simulator and output buffer must be valid";
206 return -1;
207 }
208 const Tensor2D &u = H(h)->sim->fieldSlice();
209 for (int y = 0; y < u.rows(); ++y)
210 std::memcpy(out + static_cast<size_t>(y) * u.cols(), u[y].data(), sizeof(float) * u.cols());
211 return 0;
212}
